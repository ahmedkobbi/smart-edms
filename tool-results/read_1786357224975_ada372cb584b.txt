    73→  const router = useRouter();

    93→    queryFn: () => api.get(`/api/documents/${params.id}`),

   108→    queryKey: ['classifications'],
   109→    queryFn: () => api.get('/api/classifications'),

   128→  });

   131→    mutationFn: () => api.post(`/api/documents/${params.id}/ai-suggest`),

   134→        title: 'AI suggestion ready',
   135→        description: `Suggested: ${res.suggestion.name}. Review and approve to apply.`,

   143→    mutationFn: (classificationId: string) =>

   160→    onSuccess: () => {
   161→      toast({ title: 'Version uploaded' });
   162→      setNewVersionFile(null);
   163→      setChangeReason('');
   164→      qc.invalidateQueries({ queryKey: ['document', params.id] });
   165→    },
   166→    onError: (err: any) => toast({ title: 'Upload failed', description: err?.message, variant: 'destructive' }),
   167→  });
   168→
   169→  if (isLoading || !data) {
   170→    return (
   171→      <div className="flex items-center justify-center py-12">
   172→        <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   173→      </div>
   174→    );
   175→  }
   176→
   177→  const doc = data.document;
   178→  const latestVersion = doc.versions[0];
   179→  const tags = safeParseArray(doc.tags);
   180→  const metadata = safeParseObject(doc.metadata);

   186→          <ArrowLeft className="mr-1 h-3.5 w-3.5" /> Back to documents
   187→        </Link>

   189→          <div className="flex-1 min-w-0">
   190→            <div className="flex items-center gap-2 flex-wrap mb-1">

   193→                <Badge variant="outline" className="font-mono text-xs" style={{ borderColor: doc.classification.color, color: doc.classification.color }}>
   194→                  {doc.classification.code}
   195→                </Badge>
   196→              )}
   197→              <Badge variant="secondary" className="text-xs">{STATE_LABELS[doc.state] ?? doc.state}</Badge>
   198→              {doc.isRecord && <Badge variant="outline" className="text-xs">Record</Badge>}
   199→              {doc.legalHold && (
   200→                <Badge variant="outline" className="text-xs text-red-600 border-red-300">
   201→                  <FileLock className="mr-1 h-3 w-3" /> Legal hold
   202→                </Badge>
   203→              )}
   204→              {doc.isLocked && (
   205→                <Badge variant="outline" className="text-xs text-amber-600 border-amber-300">
   206→                  <Lock className="mr-1 h-3 w-3" /> Locked

   210→            {doc.description && <p className="text-sm text-muted-foreground">{doc.description}</p>}
   211→          </div>
   212→          <div className="flex gap-2">

   225→                    api.post(`/api/documents/${params.id}/declare-record`, { reason: 'Manual declaration' })
   226→                      .then(() => { toast({ title: 'Record declared' }); qc.invalidateQueries({ queryKey: ['document', params.id] }); })
   227→                      .catch((err: any) => toast({ title: 'Failed', description: err?.message, variant: 'destructive' }));
   228→                  }
   229→                }}
   230→              >
   231→                <ShieldCheck className="mr-2 h-3.5 w-3.5" /> Declare Record
   232→              </Button>

   234→            <MoveCopyDialog docId={params.id} />

   236→              <Button
   237→                variant="outline"
   238→                size="sm"

   241→              >
   242→                {doc.isLocked ? <Unlock className="mr-2 h-3.5 w-3.5" /> : <Lock className="mr-2 h-3.5 w-3.5" />}

   244→              </Button>
   245→            )}
   246→          </div>
   247→        </div>
   248→      </div>
   249→
   250→      {/* Classification banner */}
   251→      {doc.classification && (
   252→        <div
   253→          className="px-4 py-2 rounded-md text-sm font-medium text-white flex items-center gap-2"
   254→          style={{ backgroundColor: doc.classification.color }}
   255→        >
   256→          <Shield className="h-4 w-4" />
   257→          Classification: {doc.classification.name} — {doc.classification.description ?? 'Sensitive document'}
   258→        </div>

   260→
   261→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   262→        <TabsList className="grid w-full grid-cols-2 md:grid-cols-7 lg:w-fit">
   263→          <TabsTrigger value="overview">Overview</TabsTrigger>
   264→          <TabsTrigger value="preview">Preview</TabsTrigger>
   265→          <TabsTrigger value="versions">Versions</TabsTrigger>
   266→          <TabsTrigger value="comments">Comments</TabsTrigger>
   267→          <TabsTrigger value="audit">Audit</TabsTrigger>
   268→          <TabsTrigger value="share">Share</TabsTrigger>
   269→          <TabsTrigger value="collab">Collab</TabsTrigger>
   270→          <TabsTrigger value="ai">AI</TabsTrigger>
   271→        </TabsList>
   272→
   273→        {/* Overview */}
   274→        <TabsContent value="overview" className="space-y-4">
   275→          <div className="grid grid-cols-1 md:grid-cols-3 gap-4">

   287→            </Card>
   288→          </div>
   289→
   290→          {latestVersion && (
   291→            <Card>
   292→              <CardHeader><CardTitle className="text-base">Current version (v{latestVersion.versionNumber})</CardTitle></CardHeader>
   293→              <CardContent className="space-y-3">

   311→                </div>

   313→                  <div>
   314→                    <p className="text-xs text-muted-foreground">Change reason</p>

   316→                  </div>
   317→                )}
   318→              </CardContent>
   319→            </Card>
   320→          )}
   321→
   322→          {tags.length > 0 && (
   323→            <Card>
   324→              <CardHeader><CardTitle className="text-base">Tags</CardTitle></CardHeader>
   325→              <CardContent>
   326→                <div className="flex flex-wrap gap-2">
   327→                  {tags.map((t: string) => <Badge key={t} variant="secondary">{t}</Badge>)}
   328→                </div>
   329→              </CardContent>
   330→            </Card>
   331→          )}
   332→
   333→          {Object.keys(metadata).length > 0 && (
   334→            <Card>
   335→              <CardHeader><CardTitle className="text-base">Metadata</CardTitle></CardHeader>
   336→              <CardContent>
   337→                <dl className="grid grid-cols-2 md:grid-cols-3 gap-3 text-sm">
   338→                  {Object.entries(metadata).map(([k, v]) => (
   339→                    <div key={k}>
   340→                      <dt className="text-xs text-muted-foreground">{k}</dt>
   341→                      <dd className="font-medium">{String(v)}</dd>

   343→                  ))}
   344→                </dl>
   345→              </CardContent>
   346→            </Card>

   348→

   362→
   363→          {hasPermission(perms, PERMISSIONS.DOCUMENT_CLASSIFY) && classificationsData && (

  1281→function RedactionEntry({ docId, doc }: { docId: string; doc: any }) {

  1290→      const res = await api.get<{ url: string; mimeType: string }>(`/api/documents/${docId}/preview`);

  1295→    } finally {
  1296→      setLoading(false);
  1297→    }
  1298→  }

  1300→  if (editing && previewUrl) {

  1304→        docId={docId}
  1305→        previewUrl={previewUrl}

  1307→        onClose={() => { setEditing(false); setPreviewUrl(null); }}
  1308→      />
  1309→    );

  1314→      <CardContent className="p-4 flex items-center justify-between">

  1322→        </Button>
  1323→      </CardContent>
  1324→    </Card>
