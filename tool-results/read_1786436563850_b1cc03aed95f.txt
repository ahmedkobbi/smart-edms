    75→  const { toast } = useToast();
    76→  const qc = useQueryClient();

    85→      api.get<{ url: string }>(`/api/documents/${params.id}/download`)

    94→    queryFn: () => api.get(`/api/documents/${params.id}`),

    99→    queryKey: ['document-audit', params.id],
   100→    queryFn: () => api.get(`/api/documents/${params.id}/audit`),
   101→  });

   106→  });

   108→  const { data: classificationsData } = useQuery<{ items: any[] }>({
   109→    queryKey: ['classifications'],
   110→    queryFn: () => api.get('/api/classifications'),

   129→  });

   132→    mutationFn: () => api.post(`/api/documents/${params.id}/ai-suggest`),

   134→      toast({
   135→        title: 'AI suggestion ready',
   136→        description: `Suggested: ${res.suggestion.name}. Review and approve to apply.`,
   137→      });
   138→      qc.invalidateQueries({ queryKey: ['document', params.id] });
   139→    },
   140→    onError: (err: any) => toast({ title: 'AI failed', description: err?.message, variant: 'destructive' }),
   141→  });

   144→    mutationFn: (classificationId: string) =>
   145→      api.patch(`/api/documents/${params.id}`, { classificationId, reason: 'Manual classification' }),

   147→      toast({ title: 'Classification updated' });

   167→    onError: (err: any) => toast({ title: 'Upload failed', description: err?.message, variant: 'destructive' }),
   168→  });

   170→  // Restore a previous version (creates a new forward-only version)

   172→    mutationFn: ({ versionId, reason }: { versionId: string; reason: string }) =>
   173→      api.post(`/api/documents/${params.id}/versions/${versionId}/restore`, { reason }),
   174→    onSuccess: () => {
   175→      toast({ title: 'Version restored', description: 'A new version was created with the restored content.' });
   176→      qc.invalidateQueries({ queryKey: ['document', params.id] });
   177→    },
   178→    onError: (err: any) => toast({ title: 'Restore failed', description: err?.message, variant: 'destructive' }),
   179→  });
   180→
   181→  if (isLoading || !data) {
   182→    return (
   183→      <div className="flex items-center justify-center py-12">
   184→        <Loader2 className="h-6 w-6 animate-spin text-muted-foreground" />
   185→      </div>
   186→    );
   187→  }
   188→

   190→  const latestVersion = doc.versions[0];
   191→  const tags = safeParseArray(doc.tags);

   198→          <ArrowLeft className="me-1 h-3.5 w-3.5" /> Back to documents
   199→        </Link>
   200→        <div className="flex items-start justify-between gap-4 flex-wrap">
   201→          <div className="flex-1 min-w-0">
   202→            <div className="flex items-center gap-2 flex-wrap mb-1">
   203→              <h1 className="text-2xl font-semibold tracking-tight">{doc.title}</h1>
   204→              {doc.classification && (
   205→                <Badge variant="outline" className="font-mono text-xs" style={{ borderColor: doc.classification.color, color: doc.classification.color }}>
   206→                  {doc.classification.code}
   207→                </Badge>
   208→              )}
   209→              <Badge variant="secondary" className="text-xs">{STATE_LABELS[doc.state] ?? doc.state}</Badge>
   210→              {doc.isRecord && <Badge variant="outline" className="text-xs">Record</Badge>}
   211→              {doc.legalHold && (
   212→                <Badge variant="outline" className="text-xs text-red-600 border-red-300">
   213→                  <FileLock className="me-1 h-3 w-3" /> Legal hold
   214→                </Badge>
   215→              )}
   216→              {doc.isLocked && (
   217→                <Badge variant="outline" className="text-xs text-amber-600 border-amber-300">
   218→                  <Lock className="me-1 h-3 w-3" /> Locked
   219→                </Badge>
   220→              )}
   221→            </div>
   222→            {doc.description && <p className="text-sm text-muted-foreground">{doc.description}</p>}
   223→          </div>
   224→          <div className="flex gap-2">
   225→            {hasPermission(perms, PERMISSIONS.DOCUMENT_DOWNLOAD) && doc.downloadAllowed && (
   226→              <Button variant="outline" size="sm" onClick={handleDownload}>
   227→                <Download className="me-2 h-3.5 w-3.5" /> Download
   228→              </Button>
   229→            )}
   230→            <FavoriteButton docId={params.id} />
   231→            {hasPermission(perms, PERMISSIONS.DOCUMENT_DECLARE_RECORD) && !doc.isRecord && (
   232→              <Button
   233→                variant="outline"
   234→                size="sm"
   235→                onClick={() => {
   236→                  if (confirm('Declare this document as a record? Records are immutable and can only be disposed via retention disposition.')) {
   237→                    api.post(`/api/documents/${params.id}/declare-record`, { reason: 'Manual declaration' })
   238→                      .then(() => { toast({ title: 'Record declared' }); qc.invalidateQueries({ queryKey: ['document', params.id] }); })
   239→                      .catch((err: any) => toast({ title: 'Failed', description: err?.message, variant: 'destructive' }));
   240→                  }
   241→                }}
   242→              >
   243→                <ShieldCheck className="me-2 h-3.5 w-3.5" /> Declare Record
   244→              </Button>
   245→            )}
   246→            <MoveCopyDialog docId={params.id} />
   247→            {hasPermission(perms, PERMISSIONS.DOCUMENT_LOCK) && (
   248→              <Button
   249→                variant="outline"
   250→                size="sm"
   251→                onClick={() => lockMutation.mutate(!doc.isLocked)}
   252→                disabled={lockMutation.isPending}
   253→              >
   254→                {doc.isLocked ? <Unlock className="me-2 h-3.5 w-3.5" /> : <Lock className="me-2 h-3.5 w-3.5" />}
   255→                {doc.isLocked ? 'Unlock' : 'Lock'}
   256→              </Button>
   257→            )}
   258→          </div>
   259→        </div>
   260→      </div>
   261→
   262→      {/* Classification banner */}
   263→      {doc.classification && (
   264→        <div
   265→          className="px-4 py-2 rounded-md text-sm font-medium text-white flex items-center gap-2"
   266→          style={{ backgroundColor: doc.classification.color }}
   267→        >
   268→          <Shield className="h-4 w-4" />
   269→          Classification: {doc.classification.name} — {(doc.classification as any).description ?? 'Sensitive document'}
   270→        </div>
   271→      )}
   272→
   273→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   274→        <TabsList className="grid w-full grid-cols-2 md:grid-cols-7 lg:w-fit">
   275→          <TabsTrigger value="overview">Overview</TabsTrigger>
   276→          <TabsTrigger value="preview">Preview</TabsTrigger>
   277→          <TabsTrigger value="versions">Versions</TabsTrigger>
   278→          <TabsTrigger value="comments">Comments</TabsTrigger>
   279→          <TabsTrigger value="audit">Audit</TabsTrigger>
   280→          <TabsTrigger value="share">Share</TabsTrigger>
   281→          <TabsTrigger value="collab">Collab</TabsTrigger>
   282→          <TabsTrigger value="ai">AI</TabsTrigger>
   283→        </TabsList>
   284→
   285→        {/* Overview */}

   299→            </Card>

   301→
   302→          {latestVersion && (

   305→              <CardContent className="space-y-3">

   320→                    <p className="text-xs text-muted-foreground">SHA-256</p>

   323→                </div>
   324→                {latestVersion.changeReason && (
   325→                  <div>
   326→                    <p className="text-xs text-muted-foreground">Change reason</p>
   327→                    <p className="text-sm">{latestVersion.changeReason}</p>
   328→                  </div>
   329→                )}
   330→              </CardContent>
   331→            </Card>
   332→          )}
   333→
   334→          {tags.length > 0 && (
   335→            <Card>
   336→              <CardHeader><CardTitle className="text-base">Tags</CardTitle></CardHeader>
   337→              <CardContent>
   338→                <div className="flex flex-wrap gap-2">
   339→                  {tags.map((t: string) => <Badge key={t} variant="secondary">{t}</Badge>)}
   340→                </div>
   341→              </CardContent>
   342→            </Card>
   343→          )}
   344→
   345→          {Object.keys(metadata).length > 0 && (
   346→            <Card>
   347→              <CardHeader><CardTitle className="text-base">Metadata</CardTitle></CardHeader>
   348→              <CardContent>
   349→                <dl className="grid grid-cols-2 md:grid-cols-3 gap-3 text-sm">
   350→                  {Object.entries(metadata).map(([k, v]) => (
   351→                    <div key={k}>
   352→                      <dt className="text-xs text-muted-foreground">{k}</dt>
   353→                      <dd className="font-medium">{String(v)}</dd>

   355→                  ))}
   356→                </dl>
   357→              </CardContent>
   358→            </Card>

   360→
   361→          {doc.retentionSchedule && (

   374→
   375→          {hasPermission(perms, PERMISSIONS.DOCUMENT_CLASSIFY) && classificationsData && (

   448→                    <div className="flex-1 min-w-0">

   460→                          variant="outline"
   461→                          disabled={restoreVersionMutation.isPending}

   468→                        >
   469→                          <RotateCcw className="me-1 h-3 w-3" /> Restore

   761→function safeParseArray(s: string | null | undefined): any[] {
   762→  if (!s) return [];
   763→  try { const v = JSON.parse(s); return Array.isArray(v) ? v : []; } catch { return []; }

   765→function safeParseObject(s: string | null | undefined): Record<string, unknown> {
   766→  if (!s) return {};
   767→  try { const v = JSON.parse(s); return v && typeof v === 'object' && !Array.isArray(v) ? v : {}; } catch { return {}; }

  1310→function RedactionEntry({ docId, doc }: { docId: string; doc: any }) {

  1319→      const res = await api.get<{ url: string; mimeType: string }>(`/api/documents/${docId}/preview`);
  1320→      setPreviewUrl(res.url);

  1324→    } finally {
  1325→      setLoading(false);

  1329→  if (editing && previewUrl) {

  1334→        previewUrl={previewUrl}

  1341→  return (
  1342→    <Card className="glass-card border-0">
  1343→      <CardContent className="p-4 flex items-center justify-between">

  1351→        </Button>
  1352→      </CardContent>
  1353→    </Card>

  1355→}
