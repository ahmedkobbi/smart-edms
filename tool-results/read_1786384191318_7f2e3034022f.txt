     1→     1→/**

     3→     3→ *

     5→     5→ *   1. Dual transport: SMTP (nodemailer) when SMTP_HOST is set,
     6→     6→ *      dev console sink otherwise. Single cached transporter.
     7→     7→ *   2. Templates loaded from /messages/{locale}.json via the server-side

     9→     9→ *   3. Enterprise HTML template:
    10→    10→ *        - Table-based layout (Outlook 2010-2019 compatible)
    11→    11→ *        - Inline CSS only (email-client safe — many strip <style>)
    12→    12→ *        - Dark-mode aware via [data-ogsc] / [data-colorscheme] hints
    13→    13→ *        - Preview text (preheader) for inbox list collapse
    14→    14→ *        - Brand header with logo lockup
    15→    15→ *        - Alert severity bands (info / warning / danger)
    16→    16→ *        - Primary CTA button with safe fallback link
    17→    17→ *        - Footer with privacy notice, unsubscribe, address
    18→    18→ *        - RTL-aware via dir attribute for Arabic
    19→    19→ *   4. HMAC-signed action URLs where the link grants a capability
    20→    20→ *      (invitation accept, password reset, break-glass review) so that
    21→    21→ *      a leaked token alone is insufficient.
    22→    22→ *   5. Plain-text companion for every HTML email (anti-spam, accessibility).
    23→    23→ */

    57→    57→async function getTransporter() {
    58→    58→  if (cachedTransporter) return cachedTransporter;
    59→    59→  if (!process.env.SMTP_HOST) return null;
    60→    60→
    61→    61→  const nodemailer = await import('nodemailer');
    62→    62→  const config: SmtpConfig = {
    63→    63→    host: process.env.SMTP_HOST,
    64→    64→    port: parseInt(process.env.SMTP_PORT || '587', 10),
    65→    65→    secure: process.env.SMTP_SECURE === 'true',
    66→    66→    user: process.env.SMTP_USER || '',
    67→    67→    pass: process.env.SMTP_PASS || '',
    68→    68→    from: process.env.SMTP_FROM || 'Smart EDMS <noreply@smartedms.local>',
    69→    69→  };
    70→    70→
    71→    71→  cachedTransporter = nodemailer.createTransport({
    72→    72→    host: config.host,
    73→    73→    port: config.port,
    74→    74→    secure: config.secure,
    75→    75→    auth: config.user ? { user: config.user, pass: config.pass } : undefined,
    76→    76→  });
    77→    77→
    78→    78→  return cachedTransporter;
    79→    79→}

    84→    84→

    89→    89→    console.log(`   To: ${params.to}`);
    90→    90→    console.log(`   Subject: ${params.subject}`);

    92→    92→    console.log('');
    93→    93→    return { ok: true, message: 'Email logged (dev mode — no SMTP configured)' };
    94→    94→  }
    95→    95→
    96→    96→  try {
    97→    97→    await transporter.sendMail({
    98→    98→      from,
    99→    99→      to: params.to,
   100→   100→      subject: params.subject,
   101→   101→      html: params.html,
   102→   102→      text: params.text,
   103→   103→      replyTo: params.replyTo,
   104→   104→      headers: params.listUnsubscribe
   105→   105→        ? { 'List-Unsubscribe': params.listUnsubscribe, 'List-Unsubscribe-Post': 'List-Unsubscribe=One-Click' }
   106→   106→        : undefined,
   107→   107→    });
   108→   108→    return { ok: true, message: 'Email sent' };
   109→   109→  } catch (err: any) {
   110→   110→    logger.error('email.send_failed', { to: params.to, subject: params.subject, error: err.message });
   111→   111→    return { ok: false, message: err.message };
   112→   112→  }
   113→   113→}

   119→   119→/**
   120→   120→ * Sign a token with an HMAC-SHA256 of (token + purpose).
   121→   121→ * Returns `${token}.${signature}` so the verifier can detect tampering
   122→   122→ * even if the token is otherwise opaque.
   123→   123→ *
   124→   124→ * Uses NEXTAUTH_SECRET or AUTH_SECRET (falling back to a dev-only secret
   125→   125→ * that is loud about not being safe for production).
   126→   126→ */
   127→   127→function getEmailSigningSecret(): string {
   128→   128→  const secret = process.env.NEXTAUTH_SECRET || process.env.AUTH_SECRET;
   129→   129→  if (!secret) {
   130→   130→    if (process.env.NODE_ENV === 'production') {
   131→   131→      logger.warn('email.signing_secret_missing_in_production', {});
   132→   132→    }
   133→   133→    return 'dev-only-unsafe-email-signing-secret';
   134→   134→  }
   135→   135→  return secret;
   136→   136→}
   137→   137→
   138→   138→export function signEmailToken(token: string, purpose: string): string {
   139→   139→  const secret = getEmailSigningSecret();
   140→   140→  const sig = crypto.createHmac('sha256', secret).update(`${purpose}:${token}`).digest('hex');
   141→   141→  return `${token}.${sig}`;
   142→   142→}
   143→   143→
   144→   144→export function verifyEmailToken(signedToken: string, purpose: string): string | null {
   145→   145→  if (!signedToken.includes('.')) return null;
   146→   146→  const lastDot = signedToken.lastIndexOf('.');
   147→   147→  const token = signedToken.slice(0, lastDot);
   148→   148→  const sig = signedToken.slice(lastDot + 1);
   149→   149→  const secret = getEmailSigningSecret();
   150→   150→  const expected = crypto.createHmac('sha256', secret).update(`${purpose}:${token}`).digest('hex');
   151→   151→  // Constant-time compare to prevent timing attacks
   152→   152→  if (sig.length !== expected.length) return null;
   153→   153→  try {
   154→   154→    if (crypto.timingSafeEqual(Buffer.from(sig, 'hex'), Buffer.from(expected, 'hex'))) {
   155→   155→      return token;
   156→   156→    }
   157→   157→  } catch {
   158→   158→    // Hex decode failure
   159→   159→  }
   160→   160→  return null;
   161→   161→}

   376→   376→              ${ctaHtml}
   377→   377→              ${footerNoteHtml}
   378→   378→            </td>

   675→   675→/**
   676→   676→ * Sign a URL by appending (or replacing) the `sig` query parameter.
   677→   677→ * The signature covers the URL path + query (excluding `sig`) + the purpose.
   678→   678→ * Verifiers recompute the HMAC and compare.
   679→   679→ *
   680→   680→ * Exported for use in tests and in route handlers that need to verify
   681→   681→ * signed URLs (e.g. /accept-invite, /reset-password).
   682→   682→ */
   683→   683→export function signUrl(url: string, purpose: string): string {
   684→   684→  try {
   685→   685→    const u = new URL(url);
   686→   686→    const data = `${purpose}:${u.pathname}${u.search ? '?' + u.search.replace(/[?&]sig=[^&]*/g, '').replace(/^[?&]/, '?') : ''}`;
   687→   687→    const sig = crypto.createHmac('sha256', getEmailSigningSecret()).update(data).digest('hex');
   688→   688→    u.searchParams.set('sig', sig);
   689→   689→    return u.toString();
   690→   690→  } catch {
   691→   691→    // Not a valid URL — fall back to token-style signing
   692→   692→    return signEmailToken(url, purpose);
   693→   693→  }
   694→   694→}
