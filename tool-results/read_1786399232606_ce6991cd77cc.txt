   911→/**
   912→ * Get a localized label for an audit event type.
   913→ * Falls back to English, then to the raw event code.
   914→ */
   915→export function getAuditEventLabel(eventType: string, locale: string = 'en'): string {
   916→  const labels = AUDIT_EVENT_LABELS[eventType];
   917→  if (!labels) return eventType;
   918→  return labels[locale] || labels.en || eventType;
   919→}
