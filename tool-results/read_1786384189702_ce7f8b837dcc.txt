     1→/**
     2→ * Smart EDMS — Email delivery service (enterprise-grade)
     3→ *
     4→ * Production design:
     5→ *   1. Dual transport: SMTP (nodemailer) when SMTP_HOST is set,
     6→ *      dev console sink otherwise. Single cached transporter.
     7→ *   2. Templates loaded from /messages/{locale}.json via the server-side
     8→ *      translator — supports all 5 locales (en, fr, ar, es, de).
     9→ *   3. Enterprise HTML template:
    10→ *        - Table-based layout (Outlook 2010-2019 compatible)
    11→ *        - Inline CSS only (email-client safe — many strip <style>)
    12→ *        - Dark-mode aware via [data-ogsc] / [data-colorscheme] hints
    13→ *        - Preview text (preheader) for inbox list collapse
    14→ *        - Brand header with logo lockup
    15→ *        - Alert severity bands (info / warning / danger)
    16→ *        - Primary CTA button with safe fallback link
    17→ *        - Footer with privacy notice, unsubscribe, address
    18→ *        - RTL-aware via dir attribute for Arabic
    19→ *   4. HMAC-signed action URLs where the link grants a capability
    20→ *      (invitation accept, password reset, break-glass review) so that
    21→ *      a leaked token alone is insufficient.
    22→ *   5. Plain-text companion for every HTML email (anti-spam, accessibility).
    23→ */

    55→let cachedTransporter: any = null;
    56→
    57→async function getTransporter() {
    58→  if (cachedTransporter) return cachedTransporter;
    59→  if (!process.env.SMTP_HOST) return null;
    60→
    61→  const nodemailer = await import('nodemailer');
    62→  const config: SmtpConfig = {
    63→    host: process.env.SMTP_HOST,
    64→    port: parseInt(process.env.SMTP_PORT || '587', 10),
    65→    secure: process.env.SMTP_SECURE === 'true',
    66→    user: process.env.SMTP_USER || '',
    67→    pass: process.env.SMTP_PASS || '',
    68→    from: process.env.SMTP_FROM || 'Smart EDMS <noreply@smartedms.local>',
    69→  };
    70→
    71→  cachedTransporter = nodemailer.createTransport({
    72→    host: config.host,
    73→    port: config.port,
    74→    secure: config.secure,
    75→    auth: config.user ? { user: config.user, pass: config.pass } : undefined,
    76→  });
    77→
    78→  return cachedTransporter;
    79→}

    81→export async function sendEmail(params: EmailParams): Promise<{ ok: boolean; message: string }> {
    82→  const transporter = await getTransporter();
    83→  const from = params.from || process.env.SMTP_FROM || 'Smart EDMS <noreply@smartedms.local>';
    84→
    85→  // Dev mode — log to console
    86→  if (!transporter) {
    87→    console.log('\n📧 EMAIL (dev mode — configure SMTP_HOST for delivery)');
    88→    console.log(`   From: ${from}`);
    89→    console.log(`   To: ${params.to}`);
    90→    console.log(`   Subject: ${params.subject}`);
    91→    console.log(`   Text: ${params.text.slice(0, 240)}${params.text.length > 240 ? '…' : ''}`);
    92→    console.log('');
    93→    return { ok: true, message: 'Email logged (dev mode — no SMTP configured)' };
    94→  }
    95→
    96→  try {
    97→    await transporter.sendMail({
    98→      from,
    99→      to: params.to,
   100→      subject: params.subject,
   101→      html: params.html,
   102→      text: params.text,
   103→      replyTo: params.replyTo,
   104→      headers: params.listUnsubscribe
   105→        ? { 'List-Unsubscribe': params.listUnsubscribe, 'List-Unsubscribe-Post': 'List-Unsubscribe=One-Click' }
   106→        : undefined,
   107→    });
   108→    return { ok: true, message: 'Email sent' };
   109→  } catch (err: any) {
   110→    logger.error('email.send_failed', { to: params.to, subject: params.subject, error: err.message });
   111→    return { ok: false, message: err.message };
   112→  }
   113→}

   115→// ---------------------------------------------------------------------------
   116→//  Link signing (HMAC-SHA256)
   117→// ---------------------------------------------------------------------------

   119→/**
   120→ * Sign a token with an HMAC-SHA256 of (token + purpose).
   121→ * Returns `${token}.${signature}` so the verifier can detect tampering
   122→ * even if the token is otherwise opaque.
   123→ *
   124→ * Uses NEXTAUTH_SECRET or AUTH_SECRET (falling back to a dev-only secret
   125→ * that is loud about not being safe for production).
   126→ */
   127→function getEmailSigningSecret(): string {
   128→  const secret = process.env.NEXTAUTH_SECRET || process.env.AUTH_SECRET;
   129→  if (!secret) {
   130→    if (process.env.NODE_ENV === 'production') {
   131→      logger.warn('email.signing_secret_missing_in_production', {});
   132→    }
   133→    return 'dev-only-unsafe-email-signing-secret';
   134→  }
   135→  return secret;
   136→}
   137→
   138→export function signEmailToken(token: string, purpose: string): string {
   139→  const secret = getEmailSigningSecret();
   140→  const sig = crypto.createHmac('sha256', secret).update(`${purpose}:${token}`).digest('hex');
   141→  return `${token}.${sig}`;
   142→}
   143→
   144→export function verifyEmailToken(signedToken: string, purpose: string): string | null {
   145→  if (!signedToken.includes('.')) return null;
   146→  const lastDot = signedToken.lastIndexOf('.');
   147→  const token = signedToken.slice(0, lastDot);
   148→  const sig = signedToken.slice(lastDot + 1);
   149→  const secret = getEmailSigningSecret();
   150→  const expected = crypto.createHmac('sha256', secret).update(`${purpose}:${token}`).digest('hex');
   151→  // Constant-time compare to prevent timing attacks
   152→  if (sig.length !== expected.length) return null;
   153→  try {
   154→    if (crypto.timingSafeEqual(Buffer.from(sig, 'hex'), Buffer.from(expected, 'hex'))) {
   155→      return token;
   156→    }
   157→  } catch {
   158→    // Hex decode failure
   159→  }
   160→  return null;
   161→}

   212→function alertStyles(severity: NonNullable<AlertSeverity>): { bg: string; border: string; text: string } {
   213→  switch (severity) {

   219→      return { bg: PALETTE.alertDangerBg, border: PALETTE.alertDangerBorder, text: PALETTE.alertDangerText };

   223→/**
   224→ * Build the enterprise HTML email.
   225→ *
   226→ * Layout (all inline CSS for email-client compatibility):
   227→ *   - Outer table (background slate)
   228→ *   - Centered 600px card
   229→ *   - Brand header (logo lockup + dark band)
   230→ *   - Preheader (hidden in client)
   231→ *   - Title H1
   232→ *   - Optional alert band
   233→ *   - Body paragraph(s)
   234→ *   - Optional metadata rows (label / value)
   235→ *   - Optional CTA button (table-based for Outlook)
   236→ *   - Optional fallback link
   237→ *   - Footer (privacy notice, unsubscribe, address)
   238→ */
   239→function renderEmailTemplate(p: TemplateParams): string {
   240→  const dir = p.locale === 'ar' ? 'rtl' : 'ltr';
   241→  const isRtl = dir === 'rtl';
   242→  const align = isRtl ? 'right' : 'left';
   243→  const oppositeAlign = isRtl ? 'left' : 'right';
   244→  const alertColors = p.alert?.severity ? alertStyles(p.alert.severity) : null;
   245→
   246→  // Preheader: hidden preview text shown in inbox list. Padding keeps it
   247→  // visible to the recipient when they open the email (no display:none,
   248→  // which some clients strip).
   249→  const preheaderHtml = p.preheader
   250→    ? `<div style="display:none;font-size:1px;color:#f1f5f9;line-height:1px;max-height:0;max-width:0;opacity:0;overflow:hidden;mso-hide:all;font-family:-apple-system,BlinkMacSystemFont,'Segoe UI',Roboto,sans-serif;">${escapeHtml(p.preheader)}</div>`
   251→    : '';
   252→
   253→  const alertHtml = p.alert && alertColors
   254→    ? `<table role="presentation" width="100%" cellpadding="0" cellspacing="0" style="margin:0 0 24px 0;border-radius:8px;background:${alertColors.bg};border:1px solid ${alertColors.border};border-collapse:separate;">
   255→        <tr>
   256→          <td style="padding:16px 20px;font-family:-apple-system,BlinkMacSystemFont,'Segoe UI',Roboto,sans-serif;font-size:15px;line-height:1.5;color:${alertColors.text};text-align:${align};">${p.alert.html}</td>
   257→        </tr>
   258→      </table>`
   259→    : '';
   260→
   261→  const bodyParagraphHtml = `<div style="font-family:-apple-system,BlinkMacSystemFont,'Segoe UI',Roboto,sans-serif;font-size:15px;line-height:1.65;color:${PALETTE.textSecondary};text-align:${align};">${p.bodyHtml}</div>`;

   274→        <tr>
   275→          <td style="text-align:${align};">

   394→function escapeHtml(value: string): string {
   395→  return value
   396→    .replace(/&/g, '&amp;')
   397→    .replace(/</g, '&lt;')
   398→    .replace(/>/g, '&gt;')
   399→    .replace(/"/g, '&quot;')
   400→    .replace(/'/g, '&#39;');
   401→}

   415→export async function sendInvitationEmail(opts: SendInvitationOpts): Promise<void> {
   416→  const t = await getTranslator(opts.locale);
   417→  const signedUrl = signUrl(opts.inviteUrl, 'invitation');
   418→  const subject = t('emails.invitation.subject', { tenantName: opts.tenantName });
   419→  const heading = t.raw('emails.invitation.heading', { tenantName: opts.tenantName });
   420→  const body = t.raw('emails.invitation.body', { inviterEmail: opts.inviterEmail });
   421→  const welcome = t('emails.invitation.welcome');
   422→  const ctaLabel = t('emails.invitation.cta');
   423→  const expiresNote = t('emails.invitation.expiresNote');
   424→
   425→  const html = renderEmailTemplate({
   426→    locale: t.locale,
   427→    title: t('emails.invitation.title'),
   428→    preheader: heading,
   429→    bodyHtml: `<p style="margin:0 0 16px 0;">${body}</p><p style="margin:0 0 16px 0;">${escapeHtml(welcome)}</p>`,
   430→    cta: { label: ctaLabel, href: signedUrl },
   431→    footerNote: expiresNote,
   432→    appUrl: opts.inviteUrl.replace(/\/accept-invite.*$/, ''),
   433→  });
   434→
   435→  const text = t('emails.invitation.body', { inviterEmail: opts.inviterEmail, tenantName: opts.tenantName }) + '\n\n' +
   436→    ctaLabel + ': ' + signedUrl + '\n\n' + expiresNote;
   437→
   438→  await sendEmail({ to: opts.to, subject, html, text });
   439→}

   675→/**
   676→ * Sign a URL by appending (or replacing) the `sig` query parameter.
   677→ * The signature covers the URL path + query (excluding `sig`) + the purpose.
   678→ * Verifiers recompute the HMAC and compare.
   679→ *
   680→ * Exported for use in tests and in route handlers that need to verify
   681→ * signed URLs (e.g. /accept-invite, /reset-password).
   682→ */
   683→export function signUrl(url: string, purpose: string): string {
   684→  try {
   685→    const u = new URL(url);
   686→    const data = `${purpose}:${u.pathname}${u.search ? '?' + u.search.replace(/[?&]sig=[^&]*/g, '').replace(/^[?&]/, '?') : ''}`;
   687→    const sig = crypto.createHmac('sha256', getEmailSigningSecret()).update(data).digest('hex');
   688→    u.searchParams.set('sig', sig);
   689→    return u.toString();
   690→  } catch {
   691→    // Not a valid URL — fall back to token-style signing
   692→    return signEmailToken(url, purpose);
   693→  }
   694→}
