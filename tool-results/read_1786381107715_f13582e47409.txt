     1→/**
     2→ * Smart EDMS — Enterprise-grade text extraction + OCR pipeline
     3→ *
     4→ * Architecture:
     5→ *   1. Text extraction (no OCR):
     6→ *      - text/* + JSON: read directly
     7→ *      - PDF: pdfjs-dist getTextContent() (proper font/CID/FlateDecode handling)
     8→ *      - OOXML (.docx/.xlsx/.pptx): JSZip + XML parsing (real decompression)
     9→ *      - ODF: same zip approach
    10→ *   2. OCR (when text extraction yields < threshold):
    11→ *      - Images: tesseract CLI with configurable languages
    12→ *      - Scanned PDFs: pdfjs-dist page rasterization → PNG → tesseract per page
    13→ *   3. Language detection: franc (before OCR for language selection,
    14→ *      after OCR for document language metadata)
    15→ *   4. Confidence tracking: tesseract TSV output → per-page + mean confidence
    16→ *   5. Raw text persistence: original text preserved for display/AI/exports;
    17→ *      normalized text used only for search index
    18→ *   6. Audit events: document.ocr.started/completed/failed
    19→ *   7. Per-tenant config: languages, DPI, maxPages, minConfidence
    20→ *
    21→ * OCR is opt-in per tenant via settings.features.ocr (default: true).
    22→ * Language selection is per-tenant via settings.ocr.languages (default: ['eng','ara']).
    23→ */

    39→export interface OcrConfig {
    40→  enabled: boolean;
    41→  languages: string[];      // e.g. ['eng', 'ara', 'fra']
    42→  dpi: number;              // render DPI for PDF rasterization (default 300)
    43→  maxPages: number;         // max pages to OCR (default 50)

    48→  text: string;

    52→  languages: string;             // e.g. "eng+ara"
    53→  durationMs: number;
    54→  pageCount: number;
    55→}

    63→  ocrEngine: string;
    64→  ocrLanguages: string;
    65→  ocrDurationMs: number | null;

    82→/**
    83→ * Resolve OCR config from tenant settings.
    84→ * Falls back to defaults if settings are missing or invalid.
    85→ */
    86→export async function getOcrConfig(tenantId: string): Promise<OcrConfig> {
    87→  try {
    88→    const tenant = await db.tenant.findUnique({
    89→      where: { id: tenantId },
    90→      select: { settings: true },
    91→    });
    92→    const settings = JSON.parse(tenant?.settings || '{}');
    93→    const ocrSettings = settings?.ocr ?? {};
    94→    const features = settings?.features ?? {};
    95→
    96→    return {
    97→      enabled: features.ocr !== false,
    98→      languages: Array.isArray(ocrSettings.languages) && ocrSettings.languages.length > 0
    99→        ? ocrSettings.languages
   100→        : DEFAULT_OCR_CONFIG.languages,
   101→      dpi: typeof ocrSettings.dpi === 'number' ? ocrSettings.dpi : DEFAULT_OCR_CONFIG.dpi,
   102→      maxPages: typeof ocrSettings.maxPages === 'number' ? ocrSettings.maxPages : DEFAULT_OCR_CONFIG.maxPages,
   103→      minConfidence: typeof ocrSettings.minConfidence === 'number' ? ocrSettings.minConfidence : DEFAULT_OCR_CONFIG.minConfidence,
   104→    };
   105→  } catch {
   106→    return DEFAULT_OCR_CONFIG;
   107→  }
   108→}

   114→/**
   115→ * Extract text from a document version, run OCR if needed, and persist
   116→ * to DocumentTextIndex for full-text search + AI analysis.
   117→ *
   118→ * This function is designed to be called from:
   119→ *   - Initial upload (POST /api/documents)
   120→ *   - New version upload (POST /api/documents/:id/versions)
   121→ *   - Version restore (POST /api/documents/:id/versions/:vid/restore)
   122→ *   - Manual re-OCR (POST /api/documents/:id/reindex)
   123→ *
   124→ * It updates the Document.ocrStatus field so the UI can show OCR state.
   125→ */
   126→export async function indexDocumentText(
   127→  tenantId: string,
   128→  documentId: string,
   129→  versionId: string,
   130→): Promise<TextExtractionResult> {
   131→  const version = await db.documentVersion.findFirst({
   132→    where: { id: versionId, tenantId, documentId },
   133→  });
   134→  if (!version) throw new Error('Version not found');
   135→
   136→  const storage = getFileStorage();
   137→  const buf = await storage.get(version.storageKey);
   138→  const config = await getOcrConfig(tenantId);
   139→
   140→  // Mark OCR as processing
   141→  await db.document.update({
   142→    where: { id: documentId },
   143→    data: { ocrStatus: 'processing' },
   144→  }).catch(() => {});
   145→
   146→  let result: TextExtractionResult;
   147→
   148→  try {
   149→    // --- Step 1: Text extraction (no OCR) ---
   150→    let extractedText = '';
   151→    let pageCount = 0;
   152→
   153→    if (version.mimeType.startsWith('text/') || version.mimeType === 'application/json') {
   154→      extractedText = buf.toString('utf-8').slice(0, 200_000);
   155→    } else if (version.mimeType === 'application/pdf') {
   156→      const pdfResult = await extractPdfText(buf);
   157→      extractedText = pdfResult.text;
   158→      pageCount = pdfResult.pageCount;
   159→    } else if (isOoxmlMime(version.mimeType) || isOdfMime(version.mimeType)) {
   160→      extractedText = await extractZipXmlText(buf, version.mimeType);
   161→    }
   162→
   163→    // --- Step 2: OCR (when text extraction yields < threshold) ---
   164→    let ocrResult: OcrResult | null = null;
   165→
   166→    const needsOcr = config.enabled && extractedText.length < 50 &&
   167→      (version.mimeType.startsWith('image/') || version.mimeType === 'application/pdf');
   168→
   169→    if (needsOcr) {
   170→      // Audit: OCR started
   171→      await recordAuditEvent({
   172→        tenantId,
   173→        actorId: 'system',
   174→        actorEmail: 'system@smartedms.local',
   175→        eventType: 'document.ocr.started',
   176→        action: 'create',
   177→        resourceType: 'document',
   178→        resourceId: documentId,
   179→        result: 'allow',
   180→        metadata: {
   181→          versionId,
   182→          mimeType: version.mimeType,
   183→          languages: config.languages,
   184→          dpi: config.dpi,
   185→          maxPages: config.maxPages,
   186→        },
   187→      }).catch(() => {});
   188→
   189→      const ocrStartTime = Date.now();
   190→
   191→      try {
   192→        if (version.mimeType === 'application/pdf') {
   193→          // Scanned PDF: rasterize each page with pdfjs-dist, then OCR
   194→          ocrResult = await ocrScannedPdf(buf, config);
   195→        } else if (version.mimeType.startsWith('image/')) {
   196→          // Direct image OCR
   197→          ocrResult = await ocrImage(buf, version.mimeType, config);
   198→        }
   199→
   200→        if (ocrResult && ocrResult.text.length > extractedText.length) {
   201→          extractedText = ocrResult.text;
   202→        }
   203→      } catch (err) {
   204→        logger.warn('ocr.failed', {
   205→          documentId,
   206→          versionId,
   207→          error: (err as Error).message,
   208→        });
   209→
   210→        // Audit: OCR failed
   211→        await recordAuditEvent({
   212→          tenantId,
   213→          actorId: 'system',
   214→          actorEmail: 'system@smartedms.local',
   215→          eventType: 'document.ocr.failed',
   216→          action: 'create',
   217→          resourceType: 'document',
   218→          resourceId: documentId,
   219→          result: 'deny',
   220→          reason: (err as Error).message,
   221→          metadata: { versionId, durationMs: Date.now() - ocrStartTime },
   222→        }).catch(() => {});
   223→      }
   224→    }
   225→
   226→    // --- Step 3: Language detection ---
   227→    const detectedLang = await detectLanguageAdvanced(extractedText);
   228→
   229→    // --- Step 4: Build search-optimized text (normalized) ---
   230→    const searchText = buildSearchIndex(extractedText);
   231→
   232→    // --- Step 5: Determine OCR status ---
   233→    let ocrStatus = 'skipped';
   234→    if (ocrResult) {
   235→      if (ocrResult.confidence !== null && ocrResult.confidence < config.minConfidence) {
   236→        ocrStatus = 'low_confidence';
   237→      } else {
   238→        ocrStatus = 'success';
   239→      }
   240→    } else if (needsOcr) {
   241→      ocrStatus = 'failed';
   242→    } else if (!config.enabled) {
   243→      ocrStatus = 'skipped';
   244→    } else {
   245→      ocrStatus = 'success'; // text extraction without OCR succeeded
   246→    }
   247→
   248→    // --- Step 6: Persist ---
   249→    await db.documentTextIndex.upsert({
   250→      where: { documentId },
   251→      update: {
   252→        versionId,
   253→        extractedText: searchText,
   254→        rawText: extractedText,
   255→        ocrApplied: !!ocrResult,
   256→        ocrConfidence: ocrResult?.confidence ?? null,
   257→        ocrPageConfidences: JSON.stringify(ocrResult?.pageConfidences ?? []),
   258→        ocrEngine: ocrResult?.engine ?? 'none',
   259→        ocrLanguages: ocrResult?.languages ?? '',
   260→        ocrDurationMs: ocrResult?.durationMs ?? null,
   261→        language: detectedLang,
   262→        pageCount: pageCount || ocrResult?.pageCount || 0,
   263→        indexedAt: new Date(),
   264→      },
   265→      create: {
   266→        tenantId,
   267→        documentId,
   268→        versionId,
   269→        extractedText: searchText,
   270→        rawText: extractedText,
   271→        ocrApplied: !!ocrResult,
   272→        ocrConfidence: ocrResult?.confidence ?? null,
   273→        ocrPageConfidences: JSON.stringify(ocrResult?.pageConfidences ?? []),
   274→        ocrEngine: ocrResult?.engine ?? 'none',
   275→        ocrLanguages: ocrResult?.languages ?? '',
   276→        ocrDurationMs: ocrResult?.durationMs ?? null,
   277→        language: detectedLang,
   278→        pageCount: pageCount || ocrResult?.pageCount || 0,
   279→      },
   280→    });
   281→
   282→    // Update document OCR status + language
   283→    await db.document.update({
   284→      where: { id: documentId },
   285→      data: {
   286→        ocrStatus,
   287→        documentLanguage: detectedLang,
   288→        textDirection: detectedLang === 'ar' ? 'rtl' : 'ltr',
   289→      },
   290→    }).catch(() => {});
   291→
   292→    // Audit: OCR completed
   293→    if (ocrResult) {
   294→      await recordAuditEvent({
   295→        tenantId,
   296→        actorId: 'system',
   297→        actorEmail: 'system@smartedms.local',
   298→        eventType: 'document.ocr.completed',
   299→        action: 'create',
   300→        resourceType: 'document',
   301→        resourceId: documentId,
   302→        result: 'allow',
   303→        metadata: {
   304→          versionId,
   305→          ocrApplied: true,
   306→          engine: ocrResult.engine,
   307→          languages: ocrResult.languages,
   308→          confidence: ocrResult.confidence,
   309→          pageConfidences: ocrResult.pageConfidences,
   310→          durationMs: ocrResult.durationMs,
   311→          pageCount: ocrResult.pageCount,
   312→          ocrStatus,
   313→        },
   314→      }).catch(() => {});
   315→    }
   316→
   317→    result = {
   318→      extractedText: searchText,
   319→      rawText: extractedText,
   320→      ocrApplied: !!ocrResult,
   321→      ocrConfidence: ocrResult?.confidence ?? null,
   322→      ocrPageConfidences: ocrResult?.pageConfidences ?? [],
   323→      ocrEngine: ocrResult?.engine ?? 'none',
   324→      ocrLanguages: ocrResult?.languages ?? '',
   325→      ocrDurationMs: ocrResult?.durationMs ?? null,
   326→      pageCount: pageCount || ocrResult?.pageCount || 0,
   327→      language: detectedLang,
   328→    };
   329→  } catch (err) {
   330→    // Mark as failed
   331→    await db.document.update({
   332→      where: { id: documentId },
   333→      data: { ocrStatus: 'failed' },
   334→    }).catch(() => {});
   335→    throw err;
   336→  }
   337→
   338→  // Trigger semantic embedding generation (best-effort, non-blocking)
   339→  try {
   340→    const { indexDocumentEmbedding } = await import('@/lib/search/semantic-search');
   341→    indexDocumentEmbedding(documentId).catch(() => {});
   342→  } catch {}
   343→
   344→  return result;
   345→}

   348→//  PDF text extraction (pdfjs-dist)

   351→/**
   352→ * Extract text from a PDF using pdfjs-dist.
   353→ * This handles CID fonts, FlateDecode streams, and all modern PDF features
   354→ * that the old regex-based approach couldn't handle.
   355→ */
   356→async function extractPdfText(buf: Buffer): Promise<{ text: string; pageCount: number }> {
   357→  try {
   358→    // Dynamic import — pdfjs-dist is heavy and only needed for PDFs
   359→    const pdfjs = await import('pdfjs-dist/legacy/build/pdf.mjs');
   360→
   361→    const loadingTask = pdfjs.getDocument({
   362→      data: new Uint8Array(buf),
   363→      isEvalSupported: false,
   364→    } as any);
   365→
   366→    const pdfDoc = await loadingTask.promise;
   367→    const pageCount = pdfDoc.numPages;
   368→    const maxPages = Math.min(pageCount, 100); // cap at 100 pages
   369→    const textParts: string[] = [];
   370→
   371→    for (let i = 1; i <= maxPages; i++) {
   372→      const page = await pdfDoc.getPage(i);
   373→      const textContent = await page.getTextContent();
   374→      const pageText = textContent.items
   375→        .map((item: any) => item.str || '')
   376→        .join(' ')
   377→        .trim();
   378→      if (pageText) textParts.push(pageText);
   379→    }
   380→
   381→    await pdfDoc.destroy();
   382→
   383→    return {
   384→      text: textParts.join('\n\n').slice(0, 200_000),
   385→      pageCount,
   386→    };
   387→  } catch (err) {
   388→    logger.warn('pdf.text_extraction_failed', { error: (err as Error).message });
   389→    return { text: '', pageCount: 0 };
   390→  }
   391→}

   393→// ---------------------------------------------------------------------------
   394→//  OOXML / ODF text extraction (JSZip + XML parsing)
   395→// ---------------------------------------------------------------------------
   396→
   397→function isOoxmlMime(mime: string): boolean {
   398→  return mime.includes('openxmlformats-officedocument');
   399→}
   400→
   401→function isOdfMime(mime: string): boolean {
   402→  return mime.includes('opendocument');
   403→}
   404→
   405→/**
   406→ * Extract text from OOXML (.docx, .xlsx, .pptx) or ODF files.
   407→ * These are ZIP archives containing XML files — we decompress with JSZip
   408→ * and extract text from the relevant XML elements.
   409→ */
   410→async function extractZipXmlText(buf: Buffer, mimeType: string): Promise<string> {
   411→  try {
   412→    const JSZip = (await import('jszip')).default;
   413→    const zip = await JSZip.loadAsync(buf);
   414→    const textParts: string[] = [];
   415→
   416→    // Determine which XML files to read based on file type
   417→    const isWord = mimeType.includes('wordprocessing');
   418→    const isExcel = mimeType.includes('spreadsheet');
   419→    const isPpt = mimeType.includes('presentation');
   420→    const isOdf = mimeType.includes('opendocument');
   421→
   422→    if (isWord) {
   423→      // word/document.xml contains the main text
   424→      const docXml = await zip.file('word/document.xml')?.async('text');
   425→      if (docXml) {
   426→        textParts.push(extractTextFromXml(docXml, ['w:t']));
   427→      }
   428→      // Also check headers/footers
   429→      const headerFiles = Object.keys(zip.files).filter((f) => f.match(/word\/header\d*\.xml/));
   430→      for (const hf of headerFiles) {
   431→        const xml = await zip.file(hf)?.async('text');
   432→        if (xml) textParts.push(extractTextFromXml(xml, ['w:t']));
   433→      }
   434→    } else if (isExcel) {
   435→      // xl/sharedStrings.xml has string values; xl/worksheets/sheet*.xml have cell refs
   436→      const sharedStrings = await zip.file('xl/sharedStrings.xml')?.async('text');
   437→      if (sharedStrings) {
   438→        textParts.push(extractTextFromXml(sharedStrings, ['t']));
   439→      }
   440→      // Also read worksheet cell values directly
   441→      const sheetFiles = Object.keys(zip.files).filter((f) => f.match(/xl\/worksheets\/sheet\d*\.xml/));
   442→      for (const sf of sheetFiles) {
   443→        const xml = await zip.file(sf)?.async('text');
   444→        if (xml) textParts.push(extractTextFromXml(xml, ['v', 't']));
   445→      }
   446→    } else if (isPpt) {
   447→      // ppt/slides/slide*.xml
   448→      const slideFiles = Object.keys(zip.files).filter((f) => f.match(/ppt\/slides\/slide\d*\.xml/));
   449→      for (const sf of slideFiles.slice(0, 100)) {
   450→        const xml = await zip.file(sf)?.async('text');
   451→        if (xml) textParts.push(extractTextFromXml(xml, ['a:t']));
   452→      }
   453→    } else if (isOdf) {
   454→      // content.xml is the main content
   455→      const contentXml = await zip.file('content.xml')?.async('text');
   456→      if (contentXml) {
   457→        // ODF uses text:h and text:p elements
   458→        textParts.push(extractTextFromXml(contentXml, ['text:h', 'text:p', 'table:table-cell']));
   459→      }
   460→    }
   461→
   462→    return textParts.join(' ').slice(0, 200_000);
   463→  } catch (err) {
   464→    logger.warn('ooxml.text_extraction_failed', { mimeType, error: (err as Error).message });
   465→    return '';
   466→  }
   467→}

   469→/**
   470→ * Extract text content from specified XML elements.
   471→ * Uses regex (not a full XML parser) for performance — good enough for
   472→ * text extraction where we just need the inner text of known elements.
   473→ */
   474→function extractTextFromXml(xml: string, tagNames: string[]): string {
   475→  const parts: string[] = [];
   476→  for (const tag of tagNames) {
   477→    // Match <tag>...</tag> or <tag attr="...">...</tag>
   478→    const regex = new RegExp(`<${tag}[^>]*>([^<]*)</${tag}>`, 'g');
   479→    let match;
   480→    while ((match = regex.exec(xml)) !== null) {
   481→      if (match[1] && match[1].trim()) {
   482→        parts.push(match[1].trim());
   483→      }
   484→    }
   485→  }
   486→  return parts.join(' ');
   487→}

   489→// ---------------------------------------------------------------------------
   490→//  OCR: Image
   491→// ---------------------------------------------------------------------------
   492→
   493→/**
   494→ * Run OCR on an image buffer using the system tesseract CLI.
   495→ * Returns text + confidence data from TSV output.
   496→ */
   497→async function ocrImage(
   498→  buf: Buffer,
   499→  mimeType: string,
   500→  config: OcrConfig,
   501→): Promise<OcrResult> {
   502→  const startTime = Date.now();
   503→  const ext = mimeType.split('/')[1] === 'jpeg' ? 'jpg' : (mimeType.split('/')[1] || 'png');
   504→  const langStr = config.languages.join('+');
   505→
   506→  const tmpDir = await fs.mkdtemp(path.join(os.tmpdir(), 'smart-edms-ocr-'));
   507→  const inputFile = path.join(tmpDir, `input.${ext}`);
   508→  const outputBase = path.join(tmpDir, 'output');
   509→
   510→  try {
   511→    await fs.writeFile(inputFile, buf);
   512→
   513→    // Run tesseract with TSV output for confidence data
   514→    // TSV format: level page_num block_num par_num line_num word_num left top width height conf text
   515→    await new Promise<void>((resolve, reject) => {
   516→      execFile(
   517→        'tesseract',
   518→        [inputFile, outputBase, '-l', langStr, 'tsv'],
   519→        { timeout: 120_000, maxBuffer: 50 * 1024 * 1024 },
   520→        (err) => {
   521→          if (err) reject(err);
   522→          else resolve();
   523→        },
   524→      );
   525→    });
   526→
   527→    const tsvText = await fs.readFile(`${outputBase}.tsv`, 'utf-8');
   528→    const parsed = parseTesseractTsv(tsvText);
   529→
   530→    return {
   531→      text: parsed.text.slice(0, 200_000),
   532→      confidence: parsed.meanConfidence,
   533→      pageConfidences: [parsed.meanConfidence],
   534→      engine: 'tesseract_cli',
   535→      languages: langStr,
   536→      durationMs: Date.now() - startTime,
   537→      pageCount: 1,
   538→    };
   539→  } finally {
   540→    await fs.rm(tmpDir, { recursive: true, force: true }).catch(() => {});
   541→  }
   542→}

   544→// ---------------------------------------------------------------------------
   545→//  OCR: Scanned PDF (rasterize + OCR per page)
   546→// ---------------------------------------------------------------------------
   547→
   548→/**
   549→ * OCR a scanned PDF by:
   550→ *   1. Loading the PDF with pdfjs-dist
   551→ *   2. Rendering each page to a PNG canvas at the configured DPI
   552→ *   3. Running tesseract on each page image
   553→ *   4. Concatenating per-page text + computing per-page confidence
   554→ */
   555→async function ocrScannedPdf(
   556→  buf: Buffer,
   557→  config: OcrConfig,
   558→): Promise<OcrResult> {
   559→  const startTime = Date.now();
   560→  const langStr = config.languages.join('+');
   561→
   562→  // Load PDF
   563→  const pdfjs = await import('pdfjs-dist/legacy/build/pdf.mjs');
   564→  const pdfDoc = await pdfjs.getDocument({
   565→    data: new Uint8Array(buf),
   566→    isEvalSupported: false,
   567→  } as any).promise;
   568→
   569→  const pageCount = Math.min(pdfDoc.numPages, config.maxPages);
   570→  const pageTexts: string[] = [];
   571→  const pageConfidences: number[] = [];
   572→  const scale = config.dpi / 72; // PDF default is 72 DPI
   573→
   574→  const tmpDir = await fs.mkdtemp(path.join(os.tmpdir(), 'smart-edms-ocr-pdf-'));
   575→
   576→  try {
   577→    for (let i = 1; i <= pageCount; i++) {
   578→      const page = await pdfDoc.getPage(i);
   579→      const viewport = page.getViewport({ scale });
   580→
   581→      // Render page to canvas
   582→      const canvasFactory = (pdfjs as any).canvasFactory || new (pdfjs as any).NodeCanvasFactory();
   583→      const canvasAndContext = canvasFactory.create(viewport.width, viewport.height);
   584→      const renderContext = {
   585→        canvasContext: canvasAndContext.context,
   586→        viewport,
   587→        canvasFactory,
   588→      };
   589→
   590→      await page.render(renderContext).promise;
   591→
   592→      // Get PNG buffer from canvas
   593→      const pngBuffer = canvasAndContext.canvas.toBuffer('image/png');
   594→
   595→      // OCR the page image
   596→      const pageInputFile = path.join(tmpDir, `page-${i}.png`);
   597→      const pageOutputBase = path.join(tmpDir, `page-${i}-output`);
   598→      await fs.writeFile(pageInputFile, pngBuffer);
   599→
   600→      await new Promise<void>((resolve, reject) => {
   601→        execFile(
   602→          'tesseract',
   603→          [pageInputFile, pageOutputBase, '-l', langStr, 'tsv'],
   604→          { timeout: 120_000, maxBuffer: 50 * 1024 * 1024 },
   605→          (err) => {
   606→            if (err) reject(err);
   607→            else resolve();
   608→          },
   609→        );
   610→      });
   611→
   612→      const tsvText = await fs.readFile(`${pageOutputBase}.tsv`, 'utf-8');
   613→      const parsed = parseTesseractTsv(tsvText);
   614→
   615→      pageTexts.push(parsed.text);
   616→      pageConfidences.push(parsed.meanConfidence);
   617→
   618→      // Clean up page canvas
   619→      canvasFactory.destroy(canvasAndContext);
   620→
   621→      logger.debug('ocr.pdf_page_done', {
   622→        page: i,
   623→        totalPages: pageCount,
   624→        confidence: parsed.meanConfidence,
   625→        textLength: parsed.text.length,
   626→      });
   627→    }
   628→
   629→    await pdfDoc.destroy();
   630→  } finally {
   631→    await fs.rm(tmpDir, { recursive: true, force: true }).catch(() => {});
   632→  }
   633→
   634→  // Compute mean confidence across all pages
   635→  const validConfidences = pageConfidences.filter((c) => c >= 0);
   636→  const meanConfidence = validConfidences.length > 0
   637→    ? validConfidences.reduce((a, b) => a + b, 0) / validConfidences.length
   638→    : null;
   639→
   640→  return {
   641→    text: pageTexts.join('\n\n--- Page Break ---\n\n').slice(0, 500_000),
   642→    confidence: meanConfidence,
   643→    pageConfidences,
   644→    engine: 'tesseract_cli',
   645→    languages: langStr,
   646→    durationMs: Date.now() - startTime,
   647→    pageCount,
   648→  };
   649→}

   651→// ---------------------------------------------------------------------------
   652→//  Tesseract TSV parser
   653→// ---------------------------------------------------------------------------
   654→
   655→/**
   656→ * Parse Tesseract TSV output to extract text + confidence.
   657→ * TSV format (tab-separated):
   658→ *   level  page_num  block_num  par_num  line_num  word_num  left  top  width  height  conf  text
   659→ *
   660→ * We extract:
   661→ *   - text: all word-level (level=5) text values, joined by space
   662→ *   - meanConfidence: average of word-level conf values (excluding -1)
   663→ */
   664→function parseTesseractTsv(tsv: string): { text: string; meanConfidence: number } {
   665→  const lines = tsv.trim().split('\n');
   666→  if (lines.length < 2) return { text: '', meanConfidence: -1 };
   667→
   668→  // Skip header line
   669→  const dataLines = lines.slice(1);
   670→  const words: string[] = [];
   671→  const confidences: number[] = [];
   672→
   673→  for (const line of dataLines) {
   674→    const cols = line.split('\t');
   675→    if (cols.length < 12) continue;
   676→
   677→    const level = parseInt(cols[0], 10);
   678→    if (level !== 5) continue; // word level
   679→
   680→    const conf = parseFloat(cols[10]);
   681→    const text = cols[11];
   682→
   683→    if (text && text.trim()) {
   684→      words.push(text.trim());
   685→      if (conf >= 0) confidences.push(conf);
   686→    }
   687→  }
   688→
   689→  const meanConfidence = confidences.length > 0
   690→    ? confidences.reduce((a, b) => a + b, 0) / confidences.length
   691→    : -1;
   692→
   693→  return {
   694→    text: words.join(' '),
   695→    meanConfidence,
   696→  };
   697→}

   699→// ---------------------------------------------------------------------------
   700→//  Language detection (franc)
   701→// ---------------------------------------------------------------------------
   702→
   703→/**
   704→ * Detect document language using the franc library.
   705→ * Falls back to a simple Arabic-character-ratio heuristic if franc is
   706→ * unavailable or the text is too short.
   707→ */
   708→async function detectLanguageAdvanced(text: string): Promise<string> {
   709→  if (!text || text.trim().length < 10) return 'en';
   710→
   711→  try {
   712→    // franc requires at least a few words to detect reliably
   713→    const { franc } = await import('franc');
   714→    const detected = franc(text.slice(0, 1000));
   715→
   716→    // Map franc's ISO 639-3 codes to our supported locales
   717→    const francToLocale: Record<string, string> = {
   718→      eng: 'en',
   719→      ara: 'ar',
   720→      fra: 'fr',
   721→      spa: 'es',
   722→      deu: 'de',
   723→    };
   724→
   725→    if (detected && francToLocale[detected]) {
   726→      return francToLocale[detected];
   727→    }
   728→
   729→    // If franc detected a language we don't support, fall back to heuristic
   730→    if (detected && detected !== 'und') {
   731→      // Check if it's an Arabic script language
   732→      if (/[\u0600-\u06FF]/.test(text)) return 'ar';
   733→    }
   734→  } catch {
   735→    // franc not available — fall through to heuristic
   736→  }
   737→
   738→  // Heuristic: Arabic character ratio
   739→  const arabicChars = (text.match(/[\u0600-\u06FF]/g) || []).length;
   740→  const latinChars = (text.match(/[a-zA-Z]/g) || []).length;
   741→  const total = arabicChars + latinChars;
   742→  return total === 0 ? 'en' : arabicChars / total > 0.3 ? 'ar' : 'en';
   743→}

   748→
   749→export async function searchTextIndex(
   750→  tenantId: string,
   751→  query: string,
   752→  opts: { limit?: number } = {},
   753→): Promise<{ documentId: string; snippet: string }[]> {
   754→  const results = await db.documentTextIndex.findMany({
   755→    where: {
   756→      tenantId,
   757→      extractedText: { contains: query },
   758→    },
   759→    take: opts.limit ?? 50,
   760→    select: { documentId: true, extractedText: true, rawText: true },
   761→  });
   762→
   763→  return results.map((r) => {
   764→    // Use rawText for display snippets (preserves original case)
   765→    const source = r.rawText || r.extractedText;
   766→    const idx = source.toLowerCase().indexOf(query.toLowerCase());
   767→    const start = Math.max(0, idx - 50);
   768→    const end = Math.min(source.length, idx + query.length + 100);
   769→    const snippet = source.slice(start, end);
   770→    return { documentId: r.documentId, snippet: idx >= 0 ? `…${snippet}…` : source.slice(0, 200) };
   771→  });
   772→}
